# BallSpot v1.5.5 Store Readiness Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add minimum store-readiness basics before Play Store internal testing / TestFlight: account deletion, privacy/terms/support pages, settings links, content rights docs, and a store-readiness Artisan check command.

**Architecture:** Anonymization-based account deletion (preserves leaderboard rows, revokes tokens). Minimal public Blade pages for legal info at `/privacy`, `/terms`, `/support`. ProfileScreen gains a Settings section with legal links (`Linking.openURL`) and a guarded Delete Account flow. A read-only `ballspot:store-readiness-check` Artisan command reports warnings without mutating data.

**Tech Stack:** Laravel 12 (Sanctum, Blade, Artisan), Expo 56 React Native TypeScript, SQLite (dev)

## Global Constraints

- No destructive commands (`migrate:fresh`, `git reset`, storage deletion)
- No payments, ads, chat, realtime, gambling features
- Must keep all 87 existing tests green
- TypeScript must stay at 0 errors
- No hard delete of user data that would break leaderboards
- Demo content must NOT be deleted

<?php
namespace Tests\Feature;

use App\Models\User;
use App\Models\Sport;
use App\Models\Challenge;
use App\Models\League;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Illuminate\Support\Facades\Hash;
use Tests\TestCase;

class AccountDeletionTest extends TestCase
{
    use RefreshDatabase;

    private function makeUser(): User
    {
        return User::create([
            'name'     => 'Test User',
            'username' => 'testuser',
            'email'    => 'test@example.com',
            'password' => Hash::make('password'),
        ]);
    }

    public function test_unauthenticated_cannot_delete_account(): void
    {
        $this->deleteJson('/api/account')->assertStatus(401);
    }

    public function test_authenticated_user_can_delete_account(): void
    {
        $user = $this->makeUser();
        $token = $user->createToken('mobile')->plainTextToken;

        $this->withHeader('Authorization', "Bearer {$token}")
            ->deleteJson('/api/account')
            ->assertStatus(200)
            ->assertJson(['message' => 'Your account has been deleted.']);
    }

    public function test_token_is_revoked_after_deletion(): void
    {
        $user = $this->makeUser();
        $token = $user->createToken('mobile')->plainTextToken;

        $this->withHeader('Authorization', "Bearer {$token}")
            ->deleteJson('/api/account');

        $this->withHeader('Authorization', "Bearer {$token}")
            ->getJson('/api/me')
            ->assertStatus(401);
    }

    public function test_user_data_is_anonymized_after_deletion(): void
    {
        $user  = $this->makeUser();
        $token = $user->createToken('mobile')->plainTextToken;

        $this->withHeader('Authorization', "Bearer {$token}")
            ->deleteJson('/api/account');

        $fresh = User::find($user->id);
        $this->assertEquals('Deleted User', $fresh->name);
        $this->assertStringContainsString('@ballspot.deleted', $fresh->email);
        $this->assertStringStartsWith('deleted-', $fresh->username);
        $this->assertFalse(Hash::check('password', $fresh->password));
    }

    public function test_scores_remain_after_account_deletion(): void
    {
        $sport = Sport::firstOrCreate(['slug' => 'football'], ['name' => 'Football']);
        $challenge = Challenge::create([
            'sport_id'          => $sport->id,
            'title'             => 'Test Challenge',
            'ball_x_ratio'      => 0.5,
            'ball_y_ratio'      => 0.5,
            'difficulty'        => 'easy',
            'status'            => 'active',
            'hidden_image_path' => 'challenges/hidden/test.jpg',
        ]);

        $user  = $this->makeUser();
        $token = $user->createToken('mobile')->plainTextToken;

        // create a league and guess so scores exist
        $owner = User::create([
            'name' => 'Owner', 'username' => 'owner', 'email' => 'owner@x.com',
            'password' => Hash::make('password'),
        ]);
        $league = League::create([
            'name' => 'Test League', 'code' => 'ABC123',
            'owner_id' => $owner->id, 'status' => 'active',
        ]);
        $league->members()->attach($user->id, ['joined_at' => now()]);

        $this->withHeader('Authorization', "Bearer {$token}")
            ->deleteJson('/api/account')
            ->assertStatus(200);

        // The user row still exists (anonymized), no cascade wipe
        $this->assertDatabaseHas('users', ['id' => $user->id, 'name' => 'Deleted User']);
    }
}

```php
<?php
namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Str;

class AccountController extends Controller
{
    public function delete(Request $request): JsonResponse
    {
        $user = $request->user();
        $id   = $user->id;

        $user->tokens()->delete();

        $user->update([
            'name'     => 'Deleted User',
            'email'    => "deleted-{$id}@ballspot.deleted",
            'username' => "deleted-{$id}",
            'password' => Hash::make(Str::random(32)),
        ]);

        return response()->json(['message' => 'Your account has been deleted.']);
    }
}
```
